    <script type="importmap">
      {
        "imports": {
          "three": "https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.module.js",
          "three/addons/": "https://cdn.jsdelivr.net/npm/three@0.160.0/examples/jsm/"
        }
      }
    </script>
    <script type="module">
      import * as THREE from "three";
      import { OrbitControls } from "three/addons/controls/OrbitControls.js";

      // Scene Setup
      const canvas = document.getElementById("threeCanvas");
      const scene = new THREE.Scene();
      scene.background = new THREE.Color(0x87ceeb); // Sky blue default
      scene.fog = new THREE.Fog(0x87ceeb, 30, 80);

      const camera = new THREE.PerspectiveCamera(
        45,
        canvas.clientWidth / canvas.clientHeight,
        0.1,
        1000,
      );
      const cam3DPos = new THREE.Vector3(12, 8, 14);
      const cam2DPos = new THREE.Vector3(0, 25, 0.01);
      camera.position.copy(cam3DPos);

      const renderer = new THREE.WebGLRenderer({
        canvas,
        antialias: true,
        alpha: true,
      });
      renderer.setSize(canvas.clientWidth, canvas.clientHeight);
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.shadowMap.enabled = true;
      renderer.shadowMap.type = THREE.PCFSoftShadowMap;
      renderer.toneMapping = THREE.ACESFilmicToneMapping;
      renderer.toneMappingExposure = 1.1;

      const controls = new OrbitControls(camera, canvas);
      controls.enableDamping = true;
      controls.dampingFactor = 0.05;
      controls.target.set(0, 4, 0);
      controls.maxPolarAngle = Math.PI / 2 - 0.05;
      controls.minDistance = 8;
      controls.maxDistance = 40;

      // Lighting Setup
      const hemiLight = new THREE.HemisphereLight(0xffffff, 0x888888, 0.8);
      scene.add(hemiLight);

      const keyLight = new THREE.DirectionalLight(0xffffff, 1.5);
      keyLight.position.set(10, 20, 15);
      keyLight.castShadow = true;
      keyLight.shadow.mapSize.width = 2048;
      keyLight.shadow.mapSize.height = 2048;
      keyLight.shadow.camera.near = 0.5;
      keyLight.shadow.camera.far = 60;
      keyLight.shadow.camera.left = -25;
      keyLight.shadow.camera.right = 25;
      keyLight.shadow.camera.top = 25;
      keyLight.shadow.camera.bottom = -25;
      keyLight.shadow.bias = -0.0001;
      scene.add(keyLight);

      // Ground Mesh
      const groundGeo = new THREE.PlaneGeometry(100, 100);
      const groundMat = new THREE.MeshStandardMaterial({
        color: 0x4caf50,
        roughness: 0.9,
        metalness: 0.0,
      }); // Summer grass
      const ground = new THREE.Mesh(groundGeo, groundMat);
      ground.rotation.x = -Math.PI / 2;
      ground.receiveShadow = true;
      scene.add(ground);

      // Turbine Construction
      const turbineGroup = new THREE.Group();
      scene.add(turbineGroup);

      const towerGeo = new THREE.CylinderGeometry(0.18, 0.35, 8, 32);
      const towerMat = new THREE.MeshStandardMaterial({
        color: 0xe2e8f0,
        metalness: 0.1,
        roughness: 0.7,
      });
      const tower = new THREE.Mesh(towerGeo, towerMat);
      tower.position.y = 4;
      tower.castShadow = true;
      turbineGroup.add(tower);

      const nacelleGeo = new THREE.BoxGeometry(0.7, 0.5, 1.5);
      const nacelleMat = new THREE.MeshStandardMaterial({
        color: 0xf8fafc,
        metalness: 0.5,
        roughness: 0.3,
      });
      const nacelle = new THREE.Mesh(nacelleGeo, nacelleMat);
      nacelle.position.set(0, 8.1, 0.3);
      nacelle.castShadow = true;
      turbineGroup.add(nacelle);

      const hubGeo = new THREE.SphereGeometry(0.3, 32, 32);
      const hubMat = new THREE.MeshStandardMaterial({
        color: 0x059669,
        metalness: 0.6,
        roughness: 0.2,
      });
      const hub = new THREE.Mesh(hubGeo, hubMat);
      hub.position.set(0, 8.1, 1.05);
      hub.castShadow = true;
      turbineGroup.add(hub);

      const bladeGroup = new THREE.Group();
      bladeGroup.position.set(0, 8.1, 1.05);
      turbineGroup.add(bladeGroup);

      const bladeShape = new THREE.Shape();
      bladeShape.moveTo(0, 0);
      bladeShape.bezierCurveTo(0.5, 1, 0.6, 3, 0.15, 4.8);
      bladeShape.bezierCurveTo(0, 5.1, -0.2, 5.1, -0.25, 4.8);
      bladeShape.bezierCurveTo(0.1, 3, -0.1, 1, 0, 0);
      const bladeGeo = new THREE.ExtrudeGeometry(bladeShape, {
        depth: 0.12,
        bevelEnabled: true,
        bevelThickness: 0.06,
        bevelSize: 0.06,
        bevelSegments: 4,
      });
      bladeGeo.translate(-0.1, 0, -0.06);
      bladeGeo.rotateZ(-0.3);

      const bladeMatCarbon = new THREE.MeshStandardMaterial({
        color: 0xffffff,
        metalness: 0.4,
        roughness: 0.3,
      });
      const bladeMatPlastic = new THREE.MeshStandardMaterial({
        color: 0xe2e8f0,
        metalness: 0.1,
        roughness: 0.6,
      });
      let currentBladeMat = bladeMatCarbon;

      for (let i = 0; i < 3; i++) {
        const blade = new THREE.Mesh(bladeGeo, currentBladeMat);
        blade.rotation.z = (i * 2 * Math.PI) / 3;
        blade.castShadow = true;
        bladeGroup.add(blade);
      }

      // --- ENVIRONMENT: TREES ---
      const trees = [];
      function createTree(x, z) {
        const tree = new THREE.Group();
        const trunk = new THREE.Mesh(
          new THREE.CylinderGeometry(0.2, 0.3, 2, 8),
          new THREE.MeshStandardMaterial({ color: 0x8b4513, roughness: 0.8 }),
        );
        trunk.position.y = 1;
        trunk.castShadow = true;
        tree.add(trunk);

        const foliageMat = new THREE.MeshStandardMaterial({
          color: 0x2e8b57,
          roughness: 0.7,
        });
        const foliage = new THREE.Mesh(
          new THREE.SphereGeometry(1.5, 12, 12),
          foliageMat,
        );
        foliage.position.y = 3;
        foliage.scale.set(1, 1.2, 1);
        foliage.castShadow = true;
        tree.add(foliage);

        tree.position.set(x, 0, z);
        tree.userData.foliage = foliage;
        tree.userData.basePos = tree.position.clone();
        scene.add(tree);
        trees.push(tree);
      }

      // Scatter trees avoiding the center (turbine area)
      for (let i = 0; i < 25; i++) {
        let x = (Math.random() - 0.5) * 80;
        let z = (Math.random() - 0.5) * 80;
        if (Math.sqrt(x * x + z * z) < 6) continue; // Keep center clear
        createTree(x, z);
      }

      // --- AIRFLOW PARTICLES ---
      const airflowGroup = new THREE.Group();
      scene.add(airflowGroup);
      const windParticles = [];
      const particleGeo = new THREE.SphereGeometry(0.08, 8, 8);

      for (let i = 0; i < 60; i++) {
        const particleMat = new THREE.MeshBasicMaterial({
          color: 0x10b981,
          transparent: true,
          opacity: 0.7,
        });
        const particle = new THREE.Mesh(particleGeo, particleMat);
        particle.position.set(
          -10 - Math.random() * 15,
          2 + Math.random() * 7,
          (Math.random() - 0.5) * 5,
        );
        particle.userData.speed = 1 + Math.random() * 0.5;
        airflowGroup.add(particle);
        windParticles.push(particle);
      }
      airflowGroup.visible = false;

      // --- WEATHER PARTICLES (Leaves/Snow) ---
      const weatherGroup = new THREE.Group();
      scene.add(weatherGroup);
      const weatherParticles = [];
      const weatherGeo = new THREE.PlaneGeometry(0.2, 0.2);

      for (let i = 0; i < 150; i++) {
        const particleMat = new THREE.MeshBasicMaterial({
          color: 0xffa500,
          transparent: true,
          opacity: 0.8,
          side: THREE.DoubleSide,
        });
        const particle = new THREE.Mesh(weatherGeo, particleMat);
        particle.position.set(
          (Math.random() - 0.5) * 60,
          Math.random() * 20,
          (Math.random() - 0.5) * 60,
        );
        particle.userData.speed = 1 + Math.random();
        particle.userData.rotSpeed = (Math.random() - 0.5) * 0.1;
        weatherGroup.add(particle);
        weatherParticles.push(particle);
      }

      // --- SIMULATION STATE ---
      let windSpeed = 5.0;
      let rotationSpeed = 0;
      let is3DView = true;
      let isStressVisible = false;
      let isCameraAnimating = false;
      let airDensity = 1.225; // Summer default
      let currentSeason = "summer";
      let powerHistory = new Array(60).fill(0);

      // --- UI ELEMENTS ---
      const windSlider = document.getElementById("windSlider");
      const windDisplay = document.getElementById("windDisplay");
      const powerDisplay = document.getElementById("powerDisplay");
      const effDisplay = document.getElementById("effDisplay");
      const effBar = document.getElementById("effBar");
      const densityDisplay = document.getElementById("densityDisplay");
      const graphCanvas = document.getElementById("graphCanvas");
      const graphCtx = graphCanvas.getContext("2d");

      windSlider.addEventListener("input", (e) => {
        windSpeed = parseFloat(e.target.value);
        windDisplay.textContent = windSpeed.toFixed(1);
      });

      document.getElementById("viewToggle").addEventListener("click", (e) => {
        is3DView = !is3DView;
        e.currentTarget.innerHTML = is3DView
          ? '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M2 12s3-7 10-7 10 7 10 7-3 7-10 7-10-7-10-7Z"/><circle cx="12" cy="12" r="3"/></svg> Blueprint Mode'
          : '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M21 16V8a2 2 0 0 0-1-1.73l-7-4a2 2 0 0 0-2 0l-7 4A2 2 0 0 0 3 8v8a2 2 0 0 0 1 1.73l7 4a2 2 0 0 0 2 0l7-4A2 2 0 0 0 21 16z"/></svg> 3D Simulation';
        isCameraAnimating = true;
        towerMat.wireframe = !is3DView;
        nacelleMat.wireframe = !is3DView;
        hubMat.wireframe = !is3DView;
        currentBladeMat.wireframe = !is3DView;
      });

      document.getElementById("matPlastic").addEventListener("click", () => {
        swapBladeMaterial(bladeMatPlastic);
        document.getElementById("matPlastic").classList.add("btn-active");
        document.getElementById("matCarbon").classList.remove("btn-active");
      });

      document.getElementById("matCarbon").addEventListener("click", () => {
        swapBladeMaterial(bladeMatCarbon);
        document.getElementById("matCarbon").classList.add("btn-active");
        document.getElementById("matPlastic").classList.remove("btn-active");
      });

      function swapBladeMaterial(newMat) {
        bladeGroup.children.forEach((blade) => {
          blade.material = newMat;
          blade.material.wireframe = !is3DView;
        });
        currentBladeMat = newMat;
      }

      document
        .getElementById("airflowToggle")
        .addEventListener("click", (e) => {
          airflowGroup.visible = !airflowGroup.visible;
          e.currentTarget.classList.toggle("btn-active");
        });

      document.getElementById("stressToggle").addEventListener("click", (e) => {
        isStressVisible = !isStressVisible;
        e.currentTarget.classList.toggle("btn-active");
      });

      // --- SEASON LOGIC ---
      const seasonConfigs = {
        spring: {

          ground: 0x81c784,
          leaf: 0x66bb6a,
          density: 1.225,
          weatherColor: 0xff69b4,
        },
        summer: {
          sky: 0x87ceeb,
          ground: 0x4caf50,
          leaf: 0x2e8b57,
          density: 1.184,
          weatherColor: 0xffa500,

        autumn: {
          sky: 0xffcc80,
          ground: 0xa1887f,
          leaf: 0xe65100,
          density: 1.225,
          weatherColor: 0xbf360c,

          ground: 0xf5f5f5,
          leaf: 0x9e9e9e,
          density: 1.392,

      document.querySelectorAll(".season-btn").forEach((btn) => {

      function setSeason(season) {
        currentSeason = season;
        const config = seasonConfigs[season];

        scene.background.setHex(config.sky);
        scene.fog.color.setHex(config.sky);
        groundMat.color.setHex(config.ground);
        airDensity = config.density;
        densityDisplay.textContent = airDensity.toFixed(3);

        trees.forEach((tree) => {
          tree.userData.foliage.material.color.setHex(config.leaf);
        });

        weatherParticles.forEach((p) => {
          p.material.color.setHex(config.weatherColor);
          if (season === "summer") p.visible = false;
          else p.visible = true;
        });
      }
      setSeason("summer"); // Init

      const costPanel = document.getElementById("costPanel");
      document.getElementById("exportBtn").addEventListener("click", () => {

      // --- ANIMATION LOOP ---
      let lastTime = 0;
      let lerpFactor = 0;
      let timeAccumulator = 0;

      function animate(time) {
        const delta = Math.min((time - lastTime) / 1000, 0.1);
        lastTime = time;

        // Camera Transition
        if (isCameraAnimating) {
          lerpFactor += delta * 2;
          if (lerpFactor > 1) lerpFactor = 1;
          const targetPos = is3DView ? cam3DPos : cam2DPos;
          camera.position.lerp(targetPos, 0.05);
          if (lerpFactor >= 0.95) {
            camera.position.copy(targetPos);
            isCameraAnimating = false;
            lerpFactor = 0;
          }
        }

        // Physics & Telemetry
        // P = 0.5 * rho * A * v^3 * Cp
        const basePower = Math.pow((windSpeed - 2.5) / 7.5, 2.5) * 50;
        const power = Math.max(
          0,
          basePower * (airDensity / 1.225) * (windSpeed > 2.5 ? 1 : 0),
        );

        const targetRPM = windSpeed < 3 ? 0 : (windSpeed - 2) * 12;
        rotationSpeed += (targetRPM - rotationSpeed) * 0.05;
        bladeGroup.rotation.z += (rotationSpeed / 60) * Math.PI * 2 * delta;

        let efficiency =
          windSpeed > 2.5 ? 35 * Math.exp(-Math.pow(windSpeed - 10, 2) / 8) : 0;

        // Update Telemetry UI
        powerDisplay.textContent = power.toFixed(1);
        effDisplay.textContent = efficiency.toFixed(0);
        effBar.style.width = (efficiency / 35) * 100 + "%";

        // Tree Sway based on wind
        trees.forEach((tree, index) => {
          const sway =
            Math.sin(time * 0.001 * (windSpeed / 2) + index) *
            (windSpeed * 0.015);
          tree.userData.foliage.rotation.z = sway;
          tree.userData.foliage.position.x = sway * 0.5;
        });

        // Airflow Animation
        if (airflowGroup.visible) {
          windParticles.forEach((p) => {
            p.position.x += delta * windSpeed * p.userData.speed * 1.5;
            p.position.y += Math.sin(time * 0.005 + p.position.x) * 0.005;

            if (p.position.x > 6) {
              p.position.x = -15 - Math.random() * 5;
              p.position.y = 2 + Math.random() * 7;
              p.position.z = (Math.random() - 0.5) * 5;
            }
          });
        }

        // Weather Animation (Leaves/Snow falling)
        if (currentSeason !== "summer") {
          weatherParticles.forEach((p) => {
            p.position.y -= delta * p.userData.speed * 2; // Falling speed
            p.position.x += delta * windSpeed * 0.5; // Blown by wind
            p.rotation.z += p.userData.rotSpeed;
            p.rotation.x += p.userData.rotSpeed;

            if (p.position.y < 0) {
              p.position.y = 15 + Math.random() * 5;
              p.position.x = (Math.random() - 0.5) * 60 + windSpeed * 2;
              p.position.z = (Math.random() - 0.5) * 60;
            }
          });
        }

        // Stress Visualization
        if (isStressVisible) {
          const stressLevel = Math.max(0, (windSpeed - 8) / 4);
          hubMat.color.setRGB(stressLevel, 1 - stressLevel, 0);
          hubMat.emissive.setRGB(stressLevel, 1 - stressLevel, 0);
          hubMat.emissiveIntensity = 0.3 + stressLevel * 0.5;
        } else {
          hubMat.color.setHex(0x059669);
          hubMat.emissive.setRGB(0, 0, 0);
        }

        // Update Graph every ~0.5s
        timeAccumulator += delta;
        if (timeAccumulator > 0.5) {
          timeAccumulator = 0;
          powerHistory.shift();
          powerHistory.push(power);
          drawGraph();
        }

        controls.update();
        renderer.render(scene, camera);
        requestAnimationFrame(animate);
      }

      function drawGraph() {
        const w = graphCanvas.width;
        const h = graphCanvas.height;
        graphCtx.clearRect(0, 0, w, h);

        // Background grid
        graphCtx.strokeStyle = "#e2e8f0";
        graphCtx.lineWidth = 1;
        for (let i = 0; i < 4; i++) {
          graphCtx.beginPath();
          graphCtx.moveTo(0, (h / 4) * i);
          graphCtx.lineTo(w, (h / 4) * i);
          graphCtx.stroke();
        }

        // Line
        graphCtx.strokeStyle = "#059669";
        graphCtx.lineWidth = 2;
        graphCtx.beginPath();

        powerHistory.forEach((p, i) => {
          const x = (i / (powerHistory.length - 1)) * w;
          const y = h - (p / 60) * h; // 60W max scale
          if (i === 0) graphCtx.moveTo(x, y);
          else graphCtx.lineTo(x, y);
        });
        graphCtx.stroke();

        // Fill Area
        graphCtx.lineTo(w, h);
        graphCtx.lineTo(0, h);
        graphCtx.closePath();
        graphCtx.fillStyle = "rgba(5, 150, 105, 0.1)";
        graphCtx.fill();
      }

      window.addEventListener("resize", () => {
        camera.aspect = canvas.clientWidth / canvas.clientHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(canvas.clientWidth, canvas.clientHeight);
      });

      animate(0);
    </script>
